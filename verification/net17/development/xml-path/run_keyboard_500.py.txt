"""NET-17 MMIO ABI and intact undioded switch-matrix scenarios."""
import hashlib
import json
from pathlib import Path
import subprocess
import tempfile
import xml.etree.ElementTree as ET
from run_rtl import run
from build_keyboard_500 import ROOT,OUT,VERIFY,convert


def main():
    names=['sim/hdl/hc_models.sv','sim/net03/package_models.sv','sim/net04/package_models.sv','sim/net14b/package_models.sv','hardware/keyboard_500/keyboard_500_netlist.sv','sim/net17/tb_keyboard_500.sv']
    sources=names+['tools/run_keyboard_500.py','tools/build_keyboard_500.py','verification/net15/keyboard_500.xml','references/ti/sn74ahc14.pdf']
    report={'date':'2026-11-26','iteration':'NET-17','scope':'Native digital MMIO/connector/row drivers/two-stage sampled columns, functional switch contacts and weak pull-ups; no analog Schmitt/RC/contact/CDC/debounce/physical approval',
      'sources':[{'path':n,'sha256':hashlib.sha256((ROOT/n).read_bytes()).hexdigest()} for n in sources],
      'runs':[],'negative':[],'failures':[],'production_released':False}
    def save():(VERIFY/'results.json').write_text(json.dumps(report,indent=2)+'\n')
    with tempfile.TemporaryDirectory(prefix='isa1-mmio-') as tmp:
        work=Path(tmp)
        for timed in (0,1):
            cmd=['iverilog','-g2012','-Wall','-s','tb_keyboard_500',f'-Ptb_keyboard_500.TIMED={timed}',
              '-o',str(work/'test.vvp'),*[str(ROOT/n) for n in names]]
            row={'timed':timed,'compile':run(cmd,work)}
            if not row['compile']['exit']:row['execution']=run(['vvp',str(work/'test.vvp')],work,timeout=300)
            row['passed']=row['compile']['exit']==0 and row.get('execution',{}).get('exit')==0 and 'matrix_patterns=5681 row_scans=22724' in row['execution']['stdout']
            report['runs'].append(row)
            if not row['passed']:report['failures'].append(timed)
            save();print(json.dumps(row),flush=True)
            if report['failures']:return 1
        # Exact connector-to-membrane mapping and low-only drive are acceptance
        # invariants, not just generic inventory comparisons.
        for case,ref,pin,net_name,expected in [('Row series resistor bypassed','R21','1','ROW_WIRE_0','Series row wiring R21'),('Connector rows/columns exchanged','J1','1','COL_WIRE_0','Membrane connector pin map J1'),
          ('Row driver forced high','U3','2','+5V','Row driver not low-only U3')]:
            tree=ET.parse(VERIFY/'keyboard_500.xml');root=tree.getroot();node=None
            for net in root.findall('./nets/net'):
                for candidate in net.findall('node'):
                    if candidate.get('ref')==ref and candidate.get('pin')==pin:node=candidate;net.remove(node);break
                if node is not None:break
            assert node is not None
            next(n for n in root.findall('./nets/net') if n.get('name').removeprefix('/')==net_name).append(node)
            path=work/'bad.xml';tree.write(path)
            try:convert(path,work/'bad.sv')
            except ValueError as e:message=str(e)
            else:message='No rejection'
            row={'case':case,'expected_rejection':True,'message':message,'passed':expected in message};report['negative'].append(row)
            if not row['passed']:report['failures'].append(case)
            save();print(json.dumps(row),flush=True)
    return int(bool(report['failures']))


if __name__=='__main__':raise SystemExit(main())
