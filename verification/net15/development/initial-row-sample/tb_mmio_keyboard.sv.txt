`timescale 1ns/1ps
module tb_mmio_keyboard;
 parameter integer TIMED=1;
 reg clk=0,running=1,bank_reset_n=1,access_safe=0,io_commit=0;
 reg [5:0] io_select=0;reg [7:0] write_data=0;reg [63:0] keys=0;
 always #79.445 if(running)clk=~clk;
 tri [3:0] rows;tri [15:0] cols;
 wire boot,eb;wire [1:0] vb,row;wire [7:0] read_data;
 mmio_keyboard_netlist #(.TIMED(TIMED)) dut(clk,bank_reset_n,access_safe,io_commit,io_select,write_data,rows,cols,boot,eb,vb,row,read_data);
 genvar r,c;
 generate for(r=0;r<4;r=r+1)begin:R
   for(c=0;c<16;c=c+1)begin:C
     tranif1 contact(rows[r],cols[c],keys[r*16+c]);
   end
 end endgenerate
 integer writes=0,reads=0,scans=0,patterns=0,i,j,a,b,ra,rb,ca,cb,missing;
 reg [15:0] wanted,observed;reg [7:0] low;reg [5:0] held;
 function [15:0] reference(input integer selected);
   reg [3:0] connected_rows,nr;reg [15:0] connected_cols,nc;
   integer pass,x,y;
   begin
     connected_rows=4'b1<<selected;connected_cols=0;
     for(pass=0;pass<20;pass=pass+1)begin
       nr=connected_rows;nc=connected_cols;
       for(x=0;x<4;x=x+1)for(y=0;y<16;y=y+1)if(keys[x*16+y])begin
         if(connected_rows[x])nc[y]=1;
         if(connected_cols[y])nr[x]=1;
       end
       connected_rows=nr;connected_cols=nc;
     end
     reference=~connected_cols;
   end
 endfunction
 task write_register(input integer index,input [7:0] data);
 begin
   @(negedge clk);io_select=6'b1<<index;write_data=data;io_commit=1;
   @(posedge clk);#50;io_commit=0;writes=writes+1;
 end
 endtask
 task read_register(input integer index,input [7:0] expected);
 begin
   @(negedge clk);io_select=index<0?0:6'b1<<index;
   #120;
   if(read_data!==expected)$fatal(1,"MMIO read mismatch index=%0d got=%h expected=%h",index,read_data,expected);
   reads=reads+1;
 end
 endtask
 task scan(input integer selected);
 begin
   write_register(3,selected);
   repeat(4)@(posedge clk);
   wanted=reference(selected);
   read_register(4,wanted[7:0]);read_register(5,wanted[15:8]);
   scans=scans+1;
 end
 endtask
 task scan_pattern;
 begin
   for(j=0;j<4;j=j+1)scan(j);
   patterns=patterns+1;
 end
 endtask
 initial begin
   #100;bank_reset_n=0;#1000;bank_reset_n=1;access_safe=1;
   repeat(8)@(posedge clk);
   read_register(0,0);read_register(1,0);read_register(2,0);read_register(3,8'hff);read_register(-1,8'hff);
   for(i=0;i<256;i=i+1)begin
     write_register(0,i);read_register(0,i&3);
     write_register(1,i);read_register(1,i&1);
     write_register(3,i);if(row!==i[1:0])$fatal(1,"row register mismatch");
   end
   write_register(2,0);read_register(2,0);
   write_register(2,1);read_register(2,1);
   write_register(2,0);read_register(2,1);
   write_register(4,8'hff);write_register(5,8'hff);read_register(2,1);
   keys=0;scan_pattern();
   for(i=0;i<64;i=i+1)begin @(negedge clk);keys=64'b1<<i;scan_pattern();end
   for(a=0;a<64;a=a+1)for(b=a+1;b<64;b=b+1)begin
     @(negedge clk);keys=(64'b1<<a)|(64'b1<<b);scan_pattern();
   end
   // Every rectangle, all four L-shaped triples (including cross-membrane
   // columns) and its full four-key counterpart. Raw ghosting is expected.
   for(ra=0;ra<4;ra=ra+1)for(rb=ra+1;rb<4;rb=rb+1)
    for(ca=0;ca<16;ca=ca+1)for(cb=ca+1;cb<16;cb=cb+1)begin
     for(missing=0;missing<5;missing=missing+1)begin
       @(negedge clk);keys=(64'b1<<(ra*16+ca))|(64'b1<<(ra*16+cb))|(64'b1<<(rb*16+ca))|(64'b1<<(rb*16+cb));
       if(missing<4)keys=keys&~(64'b1<<((missing<2?ra:rb)*16+(missing[0]?cb:ca)));
       scan_pattern();
     end
    end
   @(negedge clk);keys=0;
   write_register(0,3);write_register(1,1);write_register(3,2);
   held={vb,eb,row,!boot};access_safe=0;
   for(i=0;i<4;i=i+1)write_register(i,0);
   #200;
   if({vb,eb,row,!boot}!==held || rows!==4'bzzzz)$fatal(1,"unsafe write/row drive");
   // Separate bank reset can be applied after writes are inhibited; confirm
   // asynchronous register clear even with backend stopped.
   @(negedge clk);running=0;bank_reset_n=0;#1000;
   if({vb,eb,row,!boot}!==0)$fatal(1,"stopped-clock bank reset failed");
   $display("RESULT timed=%0d writes=%0d reads=%0d matrix_patterns=%0d row_scans=%0d empty=1 singles=64 pairs=2016 L_triples=2880 rectangles=720 failures=0",TIMED,writes,reads,patterns,scans);$finish;
 end
 initial begin #100000000;$fatal(1,"MMIO/matrix watchdog");end
endmodule
